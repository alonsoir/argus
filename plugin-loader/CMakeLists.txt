cmake_minimum_required(VERSION 3.16)
project(plugin_loader VERSION 1.0.0 LANGUAGES CXX C)

# ============================================================================
# C++20 STANDARD
# ============================================================================
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ============================================================================
# COMPILER FLAGS - CONTROLLED BY ROOT MAKEFILE
# ============================================================================
# ⚠️  NO HARDCODED FLAGS - All flags come from root Makefile via CMAKE_CXX_FLAGS
# ⚠️  DO NOT set CMAKE_BUILD_TYPE, optimization or sanitizer flags here

message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
message(STATUS "C++ Standard: ${CMAKE_CXX_STANDARD}")
message(STATUS "CXX Flags (from Makefile): ${CMAKE_CXX_FLAGS}")

# ============================================================================
# SOURCES
# ============================================================================
set(SOURCES
    src/plugin_loader.cpp
)

set(PUBLIC_HEADERS
    include/plugin_loader/plugin_api.h
    include/plugin_loader/plugin_loader.hpp
)


# ============================================================================
# LIBSODIUM (ADR-025 — Ed25519 plugin verification)
# NO_DEFAULT_PATH: use only libsodium compiled from source (DAY 96-97)
# ============================================================================
find_library(SODIUM_LIB sodium
    PATHS /usr/local/lib /usr/lib
    NO_DEFAULT_PATH
)
if(NOT SODIUM_LIB)
    message(FATAL_ERROR "libsodium not found — run: sudo bash tools/provision.sh full")
endif()
find_path(SODIUM_INCLUDE sodium.h
    PATHS /usr/local/include /usr/include
    NO_DEFAULT_PATH
)
message(STATUS "libsodium: ${SODIUM_LIB}")

# ============================================================================
# MLD_PLUGIN_PUBKEY_HEX (ADR-025 D7 — DEBT-PUBKEY-RUNTIME-001)
# Ed25519 public key read at cmake-time from runtime file.
# NOT a secret — correct by design. No hardcoded value, no make sync-pubkey.
# Source: /etc/ml-defender/plugins/plugin_signing.pk (PEM, generated by provision.sh)
# ============================================================================
if(NOT DEFINED MLD_PLUGIN_PUBKEY_HEX OR MLD_PLUGIN_PUBKEY_HEX STREQUAL "")
    execute_process(
        COMMAND bash /vagrant/tools/extract-pubkey-hex.sh
        OUTPUT_VARIABLE MLD_PLUGIN_PUBKEY_HEX
        OUTPUT_STRIP_TRAILING_WHITESPACE
        RESULT_VARIABLE PUBKEY_RESULT
    )
    if(NOT PUBKEY_RESULT EQUAL 0 OR MLD_PLUGIN_PUBKEY_HEX STREQUAL "")
        message(FATAL_ERROR "DEBT-PUBKEY-RUNTIME-001: cannot read pubkey from /etc/ml-defender/plugins/plugin_signing.pk — run vagrant up first")
    endif()
    message(STATUS "MLD_PLUGIN_PUBKEY_HEX: read from runtime file (DEBT-PUBKEY-RUNTIME-001)")
else()
    message(STATUS "MLD_PLUGIN_PUBKEY_HEX: injected via CMake -D")
endif()

# ============================================================================
# ARGUS_ENTERPRISE_PUBKEY_HEX — keypair vendor enterprise (offline, ADR-044)
# NOT a secret — pubkey es público por definición.
# Generado con: python3 enterprise/scripts/generate_token.py --gen-keypair
# ============================================================================
# BACKLOG-CRYPTO-VENDOR-KEY-001: ARGUS_ENTERPRISE_PUBKEY_HEX NUNCA hardcodeado aquí.
# Fuente: vault kv get -field=hex secret/argus/enterprise/vendor-pubkey
# Inyectar vía: cmake -DARGUS_ENTERPRISE_PUBKEY_HEX=<hex> ...
if(NOT DEFINED ARGUS_ENTERPRISE_PUBKEY_HEX OR "${ARGUS_ENTERPRISE_PUBKEY_HEX}" STREQUAL "")
    if(ARGUS_VAULT_ENABLED)
        message(FATAL_ERROR "BACKLOG-CRYPTO-VENDOR-KEY-001: ARGUS_ENTERPRISE_PUBKEY_HEX requerido. Inject via -DARGUS_ENTERPRISE_PUBKEY_HEX=<hex> (fuente: Vault secret/argus/enterprise/vendor-pubkey)")
    else()
        set(ARGUS_ENTERPRISE_PUBKEY_HEX "")  # community build: no se usa
    endif()
endif()
message(STATUS "ARGUS_ENTERPRISE_PUBKEY_HEX: ${ARGUS_ENTERPRISE_PUBKEY_HEX}")

# ============================================================================
# LIBRARY TARGET
# ============================================================================
add_library(plugin_loader SHARED ${SOURCES})

target_include_directories(plugin_loader
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
    PRIVATE
        $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}>
        $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/..>
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src
        ${SODIUM_INCLUDE}
)

# dl for dlopen/dlsym
target_link_libraries(plugin_loader
    PRIVATE
        ${CMAKE_DL_LIBS}
        ${SODIUM_LIB}
)

if(ARGUS_VAULT_ENABLED)
    message(STATUS "plugin-loader: ARGUS_VAULT_ENABLED=ON — enterprise token validation activa")
endif()

target_compile_definitions(plugin_loader
    PRIVATE
        MLD_PLUGIN_PUBKEY_HEX="${MLD_PLUGIN_PUBKEY_HEX}"
        ARGUS_ENTERPRISE_PUBKEY_HEX="${ARGUS_ENTERPRISE_PUBKEY_HEX}"
        $<$<BOOL:${ARGUS_VAULT_ENABLED}>:ARGUS_VAULT_ENABLED>
)
target_compile_options(plugin_loader
    PRIVATE
        -Wall
        -Wextra
        -Wpedantic
)

set_target_properties(plugin_loader PROPERTIES
    VERSION ${PROJECT_VERSION}
    SOVERSION 1
    POSITION_INDEPENDENT_CODE ON
)

# ============================================================================
# INSTALLATION
# ============================================================================
install(TARGETS plugin_loader
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)

install(FILES ${PUBLIC_HEADERS}
    DESTINATION include/plugin_loader
)

# TESTS (optional)
# ============================================================================
option(BUILD_TESTS "Build tests" OFF)  # tests/ se añade en DAY 94
if(BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

    message(STATUS "  - ${header}")
